function Nav() {
  const [open, setOpen] = useState(false);
  const [light, setLight] = useState(false);
  const button = useRef<HTMLButtonElement>(null);
  const menu = useRef<HTMLDivElement>(null);
  const navigateTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  useEffect(() => () => { if (navigateTimer.current) clearTimeout(navigateTimer.current); }, []);
  const navigateMobile = (event: MouseEvent<HTMLAnchorElement>, href: string) => {
    event.preventDefault(); setOpen(false);
    // Let the menu's scroll lock release before starting the long anchor journey.
    navigateTimer.current = setTimeout(() => {
      const target = document.querySelector<HTMLElement>(href); if (!target) return;
      history.pushState(null, '', href);
      window.scrollTo({ top: target.getBoundingClientRect().top + scrollY - 76, behavior: matchMedia('(prefers-reduced-motion: reduce)').matches ? 'instant' : 'smooth' });
      const heading = target.querySelector<HTMLElement>('h1,h2'); heading?.setAttribute('tabindex', '-1'); heading?.focus({ preventScroll: true });
    }, 60);
  };
  useEffect(() => {
    const update = () => { const rect = document.getElementById('manifesto')?.getBoundingClientRect(); setLight(!!rect && rect.top < 70 && rect.bottom > 70); };
    window.addEventListener('scroll', update, { passive: true }); update();
    return () => window.removeEventListener('scroll', update);
  }, []);
  useEffect(() => {
    if (!open) return;
    const before = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    const first = menu.current?.querySelector<HTMLAnchorElement>('a'); first?.focus();
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') { setOpen(false); button.current?.focus(); }
      if (e.key === 'Tab') {
        const items = [button.current, ...Array.from(menu.current?.querySelectorAll<HTMLAnchorElement>('a') ?? [])].filter((item): item is HTMLButtonElement | HTMLAnchorElement => !!item);
        const index = items.indexOf(document.activeElement as HTMLAnchorElement);
        e.preventDefault(); items[(index + (e.shiftKey ? -1 : 1) + items.length) % items.length]?.focus();
      }
    };
    window.addEventListener('keydown', onKey);
    return () => { document.body.style.overflow = before; window.removeEventListener('keydown', onKey); };
  }, [open]);
  return <header className={`navigation ${light && !open ? 'navigation-light' : ''} ${open ? 'menu-open' : ''}`}>
    <a href="#watch" className="wordmark" aria-label="VYREN home">VYREN</a>
    <nav className="desktop-nav" aria-label="Main navigation">{links.map(link => <a key={link.title} href={link.href}>{link.title}</a>)}</nav>
    <a className="nav-model" href="#experience">NOCTIS 01 <Arrow diagonal /></a>
    <button ref={button} className="menu-toggle" aria-expanded={open} aria-controls="mobile-menu" aria-label={open ? 'Close menu' : 'Open menu'} onClick={() => setOpen(!open)}><span /><span /></button>
    <div ref={menu} id="mobile-menu" className="mobile-menu" hidden={!open}>
      <nav aria-label="Mobile navigation">{links.map((link, i) => <a key={link.title} href={link.href} onClick={event => navigateMobile(event, link.href)}><span>0{i + 1}</span>{link.title}<Arrow diagonal /></a>)}</nav>
      <p>VYREN / NOCTIS 01<br /><span>An exploration of precision.</span></p>
    </div>
  </header>;
}
